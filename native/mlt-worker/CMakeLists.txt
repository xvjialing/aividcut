cmake_minimum_required(VERSION 3.20)
project(aividcut_mlt_worker VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MLT REQUIRED IMPORTED_TARGET mlt-framework-7>=7.30)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libavfilter libswresample libswscale)
add_executable(aividcut-mlt-worker main.cpp audio_probe.cpp source_time.cpp source_audio.cpp audio_mix.cpp image_effects.cpp)
target_link_libraries(aividcut-mlt-worker PRIVATE PkgConfig::MLT)
target_link_libraries(aividcut-mlt-worker PRIVATE PkgConfig::AV)
find_package(Qt6 COMPONENTS Gui Widgets QUIET)
if(TARGET Qt6::Widgets)
  target_link_libraries(aividcut-mlt-worker PRIVATE Qt6::Gui Qt6::Widgets)
  target_compile_definitions(aividcut-mlt-worker PRIVATE AIVIDCUT_QT_METRICS=1)
endif()
if(MSVC)
  target_compile_options(aividcut-mlt-worker PRIVATE /W4)
else()
  target_compile_options(aividcut-mlt-worker PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_compile_definitions(aividcut-mlt-worker PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
install(TARGETS aividcut-mlt-worker RUNTIME DESTINATION bin)
